cmake_minimum_required(VERSION 3.20)
project(gc_abi_probe LANGUAGES C CXX)
add_library(gc_probe SHARED probe.cpp)
target_compile_definitions(gc_probe PRIVATE GC_BUILD)
target_compile_features(gc_probe PRIVATE cxx_std_17)
add_executable(test_probe test_probe.c)
target_link_libraries(test_probe PRIVATE gc_probe)
# Assertions are the C test contract, even in Release.
if(MSVC)
  target_compile_options(test_probe PRIVATE /UNDEBUG)
else()
  target_compile_options(test_probe PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME c_abi_contract COMMAND test_probe)
